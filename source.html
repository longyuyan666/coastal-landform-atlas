    /* === 交互状态与 DOM === */
    const stage = document.getElementById("atlas-stage");
    const overview = document.getElementById("overview-layer");
    const detailLayer = document.getElementById("detail-layer");
    const hotspots = Array.from(document.querySelectorAll(".hotspot"));
    const backButton = document.getElementById("back-button");
    const zoomIn = document.getElementById("zoom-in");
    const zoomOut = document.getElementById("zoom-out");
    const motionToggle = document.getElementById("motion-toggle");
    let currentIndex = -1;
    let viewScale = 1;
    let lastTrigger = null;

    const indexStrip = document.getElementById("index-strip");
    landforms.forEach((item, index) => {
      const button = document.createElement("button");
      button.type = "button";
      button.className = "index-chip";
      button.dataset.id = item.id;
      button.innerHTML = `<span>${String(index + 1).padStart(2, "0")}</span>${item.name}`;
      button.setAttribute("aria-label", `查看${item.name}详解`);
      button.addEventListener("click", event => openLandform(index, event.currentTarget));
      indexStrip.appendChild(button);
    });

    function updateDetail(index) {
      const item = landforms[index];
      const prev = landforms[(index - 1 + landforms.length) % landforms.length];
      const next = landforms[(index + 1) % landforms.length];
      document.getElementById("diagram-container").innerHTML = renderDiagram(item.id);
      document.getElementById("detail-number").textContent = `${String(index + 1).padStart(2, "0")} / ${landforms.length}`;
      document.getElementById("detail-category").textContent = item.categoryName;
      document.getElementById("detail-title").textContent = item.name;
      document.getElementById("detail-en").textContent = item.en;
      document.getElementById("detail-intro").textContent = item.intro;
      document.getElementById("detail-formation").textContent = item.formation;
      document.getElementById("view-badge").textContent = `${item.view} · 非等比例`;
      document.getElementById("detail-tags").innerHTML = item.tags.map(tag => `<span>${tag}</span>`).join("");
      document.getElementById("prev-name").textContent = prev.name;
      document.getElementById("next-name").textContent = next.name;
      document.getElementById("toolbar-title").textContent = `${item.name} · ${item.view}`;
      document.getElementById("toolbar-subtitle").textContent = `${item.en} / 形成机制与形态识别`;
      document.title = `${item.name}｜岸境海岸地貌图谱`;
      detailLayer.setAttribute("aria-label", `${item.name}地貌详解`);
      const info = document.querySelector(".detail-info");
      info.scrollTop = 0;
      detailLayer.scrollTop = 0;
    }

    function setOverviewTransform() {
      if (currentIndex >= 0) {
        const item = landforms[currentIndex];
        const marker = hotspots.find(node => node.dataset.id === item.id);
        const x = Number(marker.dataset.x) / 16;
        const y = Number(marker.dataset.y) / 10;
        overview.style.transformOrigin = `${x}% ${y}%`;
      } else {
        overview.style.transformOrigin = "50% 50%";
        overview.style.transform = `scale(${viewScale})`;
      }
    }

    function openLandform(index, trigger) {
      if (index < 0 || index >= landforms.length) return;
      if (trigger) lastTrigger = trigger;
      currentIndex = index;
      const item = landforms[index];
      updateDetail(index);
      hotspots.forEach(node => node.classList.toggle("is-active", node.dataset.id === item.id));
      setOverviewTransform();
      stage.classList.add("is-detail");
      detailLayer.setAttribute("aria-hidden", "false");
      document.getElementById("zoom-actions").style.visibility = "hidden";
      history.replaceState({ landform: item.id }, "", `#${item.id}`);
      window.setTimeout(() => backButton.focus({ preventScroll: true }), 420);
    }

    function closeDetail() {
      if (currentIndex < 0) return;
      currentIndex = -1;
      stage.classList.remove("is-detail");
      detailLayer.setAttribute("aria-hidden", "true");
      hotspots.forEach(node => node.classList.remove("is-active"));
      document.getElementById("zoom-actions").style.visibility = "visible";
      document.getElementById("toolbar-title").textContent = "海岸全景 · 动态立体鸟瞰";
      document.getElementById("toolbar-subtitle").textContent = "潮汐持续冲刷 · 教学模型 / 非等比例";
      document.title = "岸境｜海岸地貌立体图谱";
      history.replaceState(null, "", `${location.pathname}${location.search}`);
      setOverviewTransform();
      centerMobileMap();
      if (lastTrigger) window.setTimeout(() => lastTrigger.focus({ preventScroll: true }), 350);
    }

    function updateZoomButtons() {
      zoomOut.disabled = viewScale <= 1;
      zoomIn.disabled = viewScale >= 1.6;
    }

    zoomOut.addEventListener("click", () => { viewScale = Math.max(1, +(viewScale - 0.2).toFixed(1)); setOverviewTransform(); updateZoomButtons(); });
    document.getElementById("reset-view").addEventListener("click", () => { viewScale = 1; setOverviewTransform(); updateZoomButtons(); });
    updateZoomButtons();

    function setSeaMotion(paused) {
      document.body.classList.toggle("motion-paused", paused);
      motionToggle.setAttribute("aria-pressed", String(!paused));
      motionToggle.setAttribute("aria-label", paused ? "播放海水动态" : "暂停海水动态");
      motionToggle.title = paused ? "播放海水动态" : "暂停海水动态";
      motionToggle.querySelector("span").textContent = paused ? "动效暂停" : "海水动态";
    }
    motionToggle.addEventListener("click", () => {
      setSeaMotion(motionToggle.getAttribute("aria-pressed") === "true");
    });
    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) setSeaMotion(true);

    function centerMobileMap() {
      if (window.innerWidth <= 768 && currentIndex < 0) {
        overview.scrollLeft = Math.max(0, (1000 - overview.clientWidth) / 2);
      }
    }
    centerMobileMap();
    window.addEventListener("resize", centerMobileMap);

      button.addEventListener("click", () => {
        const filter = button.dataset.filter;
        document.querySelectorAll(".filter-button").forEach(item => item.classList.toggle("is-active", item === button));
        hotspots.forEach(node => node.classList.toggle("is-muted", filter !== "all" && node.dataset.category !== filter));
        document.querySelectorAll(".index-chip").forEach(chip => {
          const item = landforms.find(entry => entry.id === chip.dataset.id);
          chip.style.opacity = filter === "all" || item.category === filter ? "1" : ".28";
        });
      });
    });
